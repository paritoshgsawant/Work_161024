import "./styles.css";
import { useState, useEffect } from "react";

export default function App() {
  const [time, setTime] = useState(new Date());

  useEffect(() => {
    const xyz = setInterval(() => {
      setTime(new Date());
    }, 1000);

    return () => clearInterval(xyz);
  }, []);

  return (
    <div className="App">
      <h1>{time.toLocaleTimeString()}</h1>
    </div>
  );
}
